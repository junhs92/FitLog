// Log difficulty feedback for an exercise
Future<SessionExerciseFeedbackModel> logExerciseFeedback({
  required String sessionExerciseId,
  required String exerciseId,
  required ExerciseFeedback feedback,
  String? notes,
}) async {
  final data = await _client.from('session_exercise_feedback').insert({
    'session_exercise_id': sessionExerciseId,
    'exercise_id': exerciseId,
    'feedback': feedback.id,
    'notes': notes,
  }).select().single();

  return SessionExerciseFeedbackModel.fromJson(data);
}

// Get feedback history for an exercise
Future<List<SessionExerciseFeedbackModel>> getExerciseFeedbackHistory({
  required String exerciseId,
  required String clientId,
  Duration timeWindow = const Duration(days: 30),
}) async {
  final sinceDate = DateTime.now().subtract(timeWindow).toUtc().toIso8601String();

  final data = await _client
    .from('session_exercise_feedback')
    .select('''
      id, session_exercise_id, exercise_id, feedback, notes, created_at
    ''')
    .eq('exercise_id', exerciseId)
    .filter('created_at', 'gte', sinceDate)
    .order('created_at', ascending: false);

  return (data as List)
    .map((json) => SessionExerciseFeedbackModel.fromJson(json))
    .toList();
}

// Get feedback summary for a specific exercise in a client's training
Future<Map<ExerciseFeedback, int>> getFeedbackSummary({
  required String exerciseId,
  required String clientId,
}) async {
  final feedback = await getExerciseFeedbackHistory(
    exerciseId: exerciseId,
    clientId: clientId,
  );

  // Count occurrences of each feedback type
  final summary = <ExerciseFeedback, int>{};
  for (final item in feedback) {
    summary[item.feedback] = (summary[item.feedback] ?? 0) + 1;
  }

  return summary;
}
```

// Log exercise swap
Future<ExerciseSwapHistoryModel> logExerciseSwap({
  required String clientId,
  required String trainerId,
  required String originalExerciseId,
  required String replacementExerciseId,
  String? sessionId,
  ExerciseFeedback? feedbackReason,
  String? customReason,
}) async {
  final data = await _client.from('exercise_swap_history').insert({
    'client_id': clientId,
    'trainer_id': trainerId,
    'session_id': sessionId,
    'original_exercise_id': originalExerciseId,
    'replacement_exercise_id': replacementExerciseId,
    'feedback_reason': feedbackReason?.id,
    'custom_reason': customReason,
  }).select().single();

  return ExerciseSwapHistoryModel.fromJson(data);
}

// Get swap history for client
Future<List<ExerciseSwapHistoryModel>> getClientSwapHistory({
  required String clientId,
  Duration timeWindow = const Duration(days: 90),
}) async {
  final sinceDate = DateTime.now().subtract(timeWindow).toUtc().toIso8601String();

  final data = await _client
    .from('exercise_swap_history')
    .select('''
      id, client_id, trainer_id, session_id,
      original_exercise_id, replacement_exercise_id,
      feedback_reason, custom_reason, swapped_at,
      exercises!original_exercise_id(name_ko, name),
      exercises!replacement_exercise_id(name_ko, name)
    ''')
    .eq('client_id', clientId)
    .filter('swapped_at', 'gte', sinceDate)
    .order('swapped_at', ascending: false);

  return (data as List)
    .map((json) => ExerciseSwapHistoryModel.fromJson(json))
    .toList();
}

// Find exercises that are frequently swapped OUT (problematic)
Future<List<ProblematicExercise>> getProblematicExercises({
  required String clientId,
  int minSwapCount = 2,
}) async {
  final swaps = await getClientSwapHistory(clientId: clientId);

  // Count how many times each exercise was swapped OUT
  final problemCounts = <String, int>{};
  for (final swap in swaps) {
    problemCounts[swap.originalExerciseId] =
      (problemCounts[swap.originalExerciseId] ?? 0) + 1;
  }

  // Filter to problematic ones
  return problemCounts.entries
    .where((e) => e.value >= minSwapCount)
    .map((e) => ProblematicExercise(exerciseId: e.key, swapCount: e.value))
    .toList();
}

// Get swap alternatives (e.g., what does client swap Barbell Squat FOR?)
Future<Map<String, int>> getSwapAlternatives({
  required String originalExerciseId,
  required String clientId,
}) async {
  final swaps = await getClientSwapHistory(clientId: clientId);

  // Filter swaps for this exercise
  final relevantSwaps = swaps
    .where((s) => s.originalExerciseId == originalExerciseId)
    .toList();

  // Count which exercises are used as replacements
  final alternatives = <String, int>{};
  for (final swap in relevantSwaps) {
    alternatives[swap.replacementExerciseId] =
      (alternatives[swap.replacementExerciseId] ?? 0) + 1;
  }

  return alternatives;
}
```

SELECT
  esh.id,
  e1.name_ko AS original_exercise,
  e2.name_ko AS replacement_exercise,
  esh.feedback_reason,
  esh.custom_reason,
  COUNT(*) OVER (PARTITION BY esh.original_exercise_id) AS total_swaps
FROM exercise_swap_history esh
JOIN exercises e1 ON e1.id = esh.original_exercise_id
JOIN exercises e2 ON e2.id = esh.replacement_exercise_id
